  # Run the tests once to collect the C++ code generated by them, then compile it into the test executable
  add_executable(
    codegen_test_generator
    cpp/generator_main.cpp
    codegentestutils.cpp
    codegentestutils.h
    ${CODEBUILDER_TEST_SOURCES}
  )

  target_link_libraries(
    codegen_test_generator
    GTest::gtest
    GTest::gmock
    scratchcpp
    scratchcpp_mocks
    codegen_test_functions
  )

  if (MSVC)
    target_compile_options(codegen_test_generator PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/cpp/generator_asserts.h)
  else()
    target_compile_options(codegen_test_generator PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/cpp/generator_asserts.h)
  endif()

  set(CODEGEN_TEST_GENERATED_SOURCES "")

  foreach(i RANGE 15)
    list(APPEND CODEGEN_TEST_GENERATED_SOURCES ${CMAKE_CURRENT_BINARY_DIR}/generated_code_${i}.cpp)
  endforeach()

  add_custom_command(
    OUTPUT ${CODEGEN_TEST_GENERATED_SOURCES}
    COMMAND codegen_test_generator ${CODEGEN_TEST_GENERATED_SOURCES}
    DEPENDS codegen_test_generator
    COMMENT "Generating C++ code for codegen tests"
  )

  target_sources(codegen_test PRIVATE ${CODEGEN_TEST_GENERATED_SOURCES})
endif()
